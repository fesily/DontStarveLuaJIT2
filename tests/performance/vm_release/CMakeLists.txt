cmake_minimum_required(VERSION 3.20)
project(game_performance_vm C ASM)
enable_testing()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Performance VMs must use CMAKE_BUILD_TYPE=Release")
endif()
set(LUAJIT_TEST_RETM OFF CACHE BOOL "" FORCE)
set(LUAJIT_TEST_STACK_OOM OFF CACHE BOOL "" FORCE)
set(LUAJIT_ENABLE_FSANITIZE OFF CACHE BOOL "" FORCE)
if(MSVC)
    add_compile_options("/FI${CMAKE_CURRENT_LIST_DIR}/release_guard.h")
else()
    add_compile_options(-include "${CMAKE_CURRENT_LIST_DIR}/release_guard.h")
endif()
add_subdirectory("${DS_PERF_LUAJIT_SOURCE}" luajit)
if(MSVC)
    target_link_options(luajit-5.1 PRIVATE "/MAP:$<TARGET_FILE:luajit-5.1>.map")
endif()
